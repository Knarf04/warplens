// Which structure makes the electing half run before the other half? before/after flags.
#include "../common.h"
#define ELECT() asm volatile("{\n.reg .pred p;\nelect.sync %0|p, %2;\nselp.u32 %1, 1, 0, p;\n}" : "=r"(leader), "=r"(chosen) : "r"(mask))
extern "C" __global__ void v(uint32_t mask, int variant, int low, int* flag, uint32_t* out) {
    uint32_t lane = threadIdx.x, leader = 0xEEu, chosen = 0xEEu, before = 0xEEu, after = 0xEEu;
    const bool electing = low ? lane < 16 : lane >= 16;
    if (variant == 0) {
        if (electing) { before = flag[lane ^ 16]; ELECT(); after = flag[lane ^ 16]; }
        else { flag[lane] = 1; }
    } else if (variant == 1) {
        if (!electing) { flag[lane] = 1; }
        else { before = flag[lane ^ 16]; ELECT(); after = flag[lane ^ 16]; }
    } else if (variant == 2) {   // the other half returns early after its store
        if (!electing) { flag[lane] = 1; out[lane * 4] = 0xAA; return; }
        before = flag[lane ^ 16]; ELECT(); after = flag[lane ^ 16];
    } else {                     // the other half spins first, so its store comes late
        if (!electing) { volatile int x = 0; for (int i = 0; i < 2000; ++i) x += i; flag[lane] = 1 + (x & 0); }
        else { before = flag[lane ^ 16]; ELECT(); after = flag[lane ^ 16]; }
    }
    out[lane * 4] = leader; out[lane * 4 + 1] = chosen; out[lane * 4 + 2] = before; out[lane * 4 + 3] = after;
}
int main(int argc, char** argv) {
    int variant = atoi(argv[1]), low = atoi(argv[2]), full = atoi(argv[3]);
    uint32_t mask = full ? 0xFFFFFFFFu : (low ? 0x0000FFFFu : 0xFFFF0000u);
    uint32_t out[128]; int flag[32] = {};
    uint32_t* d_out = dev_alloc<uint32_t>(128); int* d_flag = dev_copy(flag, 32);
    dev_fill(d_out, 128, 0);
    v<<<1, 32>>>(mask, variant, low, d_flag, d_out);
    probe_sync("elect order");
    dev_read(out, d_out, 128);
    int l = low ? 0 : 16;
    printf("variant %d low %d mask %08x: leader %u before", variant, low, mask, out[l * 4]);
    for (int i = l; i < l + 16; ++i) printf(" %x", out[i * 4 + 2]);
    printf(" after");
    for (int i = l; i < l + 16; ++i) printf(" %x", out[i * 4 + 3]);
    printf("\n");
    return 0;
}
